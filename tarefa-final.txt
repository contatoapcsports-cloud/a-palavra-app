/* Chamadas às APIs gratuitas (Groq, Gemini, Cloudflare Workers AI). Sem dependências: usa fetch do Node 22.
   Regras: nunca imprime chave; 429 de limite por minuto espera e tenta de novo; 429 de limite diário bloqueia o provedor até o fim da execução. */
export class QuotaError extends Error { constructor(prov, msg) { super(prov + ': limite diário atingido (' + msg + ')'); this.quota = true; this.prov = prov; } }
export class ProvError extends Error { constructor(prov, msg, status) { super(prov + ': ' + msg); this.prov = prov; this.status = status; } }

const dormir = ms => new Promise(r => setTimeout(r, ms));
export function extrairJson(texto) {
  let t = String(texto == null ? '' : texto).trim();
  t = t.replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/, '');
  try { return JSON.parse(t); } catch { /* tenta recortar */ }
  const a = t.indexOf('{'), b = t.lastIndexOf('}');
  if (a >= 0 && b > a) { try { return JSON.parse(t.slice(a, b + 1)); } catch { /* falhou */ } }
  throw new Error('resposta sem JSON válido');
}
const DIARIO = /per day|daily|rpd|tpd|neurons|free allocation|quota exceeded|exceeded your current quota|resource_exhausted.*(day|quota)/i;
function esperaDe(resp, corpo) {
  const h = Number(resp.headers.get('retry-after'));
  if (h > 0) return h;
  const m = /try again in\s+(?:(\d+)h)?(?:(\d+)m)?(?:([\d.]+)s)?/i.exec(corpo) || /retryDelay"?\s*[:=]\s*"?(\d+)s/i.exec(corpo);
  if (m && m.length === 4) return (+m[1] || 0) * 3600 + (+m[2] || 0) * 60 + (+m[3] || 0);
  if (m && m.length === 2) return +m[1];
  return 0;
}

export class Provedor {
  /* opts: { nome, familia, modelo, base, minIntervaloMs, log, semEspera } */
  constructor(o) { Object.assign(this, { chamadas: 0, tokensIn: 0, tokensOut: 0, bloqueado: null, ultimo: 0, janela: [] }, o); }
  /* limite de tokens por minuto (TPM): espera a janela de 60 s ter espaço antes de chamar, em vez de bater no 429 */
  async reservar(est) {
    if (!this.limiteTpm || this.semEspera) return null;
    for (;;) {
      const agora = Date.now(); this.janela = this.janela.filter(x => agora - x.t < 60000);
      const soma = this.janela.reduce((a, x) => a + x.n, 0);
      if (soma + est <= this.limiteTpm || !this.janela.length) { const r = { t: agora, n: est }; this.janela.push(r); return r; }
      await dormir(Math.max(500, 60000 - (agora - this.janela[0].t) + 200));
    }
  }
  async ritmo() { // serializa o espaçamento entre chamadas, mesmo com vários trabalhos em paralelo
    const anterior = this.vez || Promise.resolve(); let solta; this.vez = new Promise(r => { solta = r; });
    await anterior;
    const falta = this.ultimo + (this.minIntervaloMs || 0) - Date.now();
    if (falta > 0 && !this.semEspera) await dormir(falta);
    this.ultimo = Date.now(); solta();
  }
  async http(url, init) {
    if (this.bloqueado) throw new QuotaError(this.nome, this.bloqueado);
    let ultimoErro;
    for (let t = 1; t <= 8; t++) {
      await this.ritmo();
      let r, corpo;
      try { r = await fetch(url, init); corpo = await r.text(); }
      catch (e) { ultimoErro = new ProvError(this.nome, 'rede: ' + e.message); if (!this.semEspera) await dormir(Math.min(5000 * t, 30000)); continue; }
      if (r.ok) { this.chamadas++; return corpo; }
      const trecho = corpo.slice(0, 400).replace(/\s+/g, ' ');
      if (r.status === 429 || (r.status === 403 && DIARIO.test(corpo))) {
        const espera = esperaDe(r, corpo);
        if (DIARIO.test(corpo) && !/per minute|rpm|tpm/i.test(corpo) || espera > 180) { this.bloqueado = trecho; throw new QuotaError(this.nome, trecho); }
        ultimoErro = new ProvError(this.nome, '429 ' + trecho, 429);
        if (!this.semEspera) await dormir(Math.min(Math.max(espera * 1000 + 1000, 3000), 120000));
        continue;
      }
      if (r.status >= 500) { ultimoErro = new ProvError(this.nome, r.status + ' ' + trecho, r.status); if (!this.semEspera) await dormir(Math.min(8000 * t, 40000)); continue; }
      throw new ProvError(this.nome, r.status + ' ' + trecho, r.status);
    }
    throw ultimoErro || new ProvError(this.nome, 'falhou');
  }
  conta(i, o) { this.tokensIn += i || 0; this.tokensOut += o || 0; }
  resumo() { return this.nome + ' (' + this.modelo + '): ' + this.chamadas + ' chamadas, ' + this.tokensIn + ' tokens entrada, ' + this.tokensOut + ' saída' + (this.bloqueado ? ' [LIMITE DIÁRIO ATINGIDO]' : ''); }
}

export class Groq extends Provedor {
  constructor(o) { super({ nome: 'groq', base: 'https://api.groq.com', minIntervaloMs: 2600, ...o }); }
  async json({ system, user, max = 4000, temp = 0.6 }) {
    const corpo = { model: this.modelo, messages: [{ role: 'system', content: system }, { role: 'user', content: user }], temperature: temp, max_completion_tokens: max, response_format: { type: 'json_object' } };
    if (this.esforco) corpo.reasoning_effort = this.esforco;
    else if (this.familia !== 'openai') corpo.reasoning_format = 'hidden'; // modelos que "pensam" (Qwen): não devolver o raciocínio no texto
    const chamar = c => this.http(this.base + '/openai/v1/chat/completions', { method: 'POST', headers: { 'content-type': 'application/json', authorization: 'Bearer ' + this.chave }, body: JSON.stringify(c) });
    const reserva = await this.reservar(Math.round((system.length + user.length) / 3.2 + 2300));
    let txt;
    for (const campo of ['', 'reasoning_format', 'reasoning_effort', 'response_format']) { // se a API recusar (400) um parâmetro opcional, tenta sem ele
      if (campo) delete corpo[campo];
      try { txt = await chamar(corpo); break; }
      catch (e) { if (!(e.status === 400 && campo !== 'response_format')) throw e; }
    }
    if (txt === undefined) throw new ProvError(this.nome, 'a API recusou o pedido (400) mesmo sem parâmetros opcionais', 400);
    const j = JSON.parse(txt), ch = j.choices && j.choices[0];
    this.conta(j.usage && j.usage.prompt_tokens, j.usage && j.usage.completion_tokens);
    if (reserva && j.usage) reserva.n = (j.usage.prompt_tokens || 0) + (j.usage.completion_tokens || 0);
    if (!ch || !ch.message || !ch.message.content) throw new ProvError(this.nome, 'resposta vazia (' + (ch && ch.finish_reason) + ')');
    if (ch.finish_reason === 'length') throw new ProvError(this.nome, 'resposta cortada (max tokens)');
    return extrairJson(ch.message.content);
  }
}

const gens = cfg.geradores.map(d => obter(d.id)), aud1s = cfg.auditores_1.map(d => obter(d.id)), aud2s = cfg.auditores_2.map(d => obter(d.id));

function escolher() {
  for (const g of gens) {
    if (g.bloqueado) continue;
    const a1 = aud1s.find(a => !a.bloqueado && a.familia !== g.familia); if (!a1) continue;
    const a2 = aud2s.find(a => !a.bloqueado && a.familia !== g.familia && a.familia !== a1.familia); if (!a2) continue;

/* ---------- montagem do ensino ---------- */
function rng(seed) { let s = 0; for (const c of seed) s = (s * 31 + c.charCodeAt(0)) >>> 0; return () => { s = (s * 1664525 + 1013904223) >>> 0; return s / 4294967296; }; }
function montar(g, u) {
  const t = g.teste || {}, rand = rng(u.id), erradas = Array.isArray(t.erradas) ? t.erradas.slice(0, 2) : [];
  const ops = [String(t.certa || ''), ...erradas.map(String)];
  const ordem = ops.map((_, i) => i).sort(() => rand() - 0.5);
  const opcoes = ordem.map(i => ops[i]), certa = ordem.indexOf(0);
  const refs = refsValidas(Array.isArray(g.refs) ? g.refs : [], existe, u).map(x => x.split('|')[1]);
  return {
    id: u.id, ref: u.ref, book: u.book, chapter: u.chapter, verses: u.verses,
    titulo: String(g.titulo || '').trim(), temas: (Array.isArray(g.temas) ? g.temas : []).map(x => String(x).trim().toLowerCase()),
    pergunta: String(g.pergunta || '').trim(), entender: String(g.entender || '').trim(),
    ligar: { texto: String(g.ligar || '').trim(), selo: g.ligar_selo === true },
    viver: String(g.viver || '').trim(), guarde: String(g.guarde || '').trim(), ore: String(g.ore || '').trim(),
    quiz: { pergunta: String(t.pergunta || '').trim(), opcoes, certa, explicacao: String(t.explicacao || '').trim() },
    refs
  };
}
function lerAuditoria(r) {
  const c = r && (r.criterios || r.gates || r.gate_results);
  if (!c || typeof c !== 'object') return { erro: 'sem o objeto criterios' };
  const falhas = [], faltam = [];
  for (const g of GATES) {
    const v = String(c[g] == null ? '' : c[g]).trim().toUpperCase();
    if (v === 'P' || v === 'PASS' || v === 'PASSOU') continue;
    if (v === 'F' || v === 'FAIL' || v === 'FALHOU') falhas.push(g); else faltam.push(g);
  }
  if (faltam.length) return { erro: 'criterios sem P/F: ' + faltam.slice(0, 5).join(',') };
  const motivos = new Map(); (Array.isArray(r.problemas) ? r.problemas : []).forEach(p => { if (p && p.criterio) motivos.set(String(p.criterio), String(p.motivo || '')); });
  return { falhas: falhas.map(g => g + (motivos.get(g) ? ': ' + motivos.get(g) : '')), aprovado: falhas.length === 0 };
}
async function auditar(prov, u, d) {
  for (let t = 1; t <= 2; t++) {
    const r = await prov.json({ system: SISTEMA_AUDITOR(doutrina), user: pedidoAuditor(u, d), max: prov.tipo === 'gemini' ? 8192 : 4000, temp: 0.1 });
    const a = lerAuditoria(r);
    if (!a.erro) return a;
    log('    [' + u.id + '] ' + prov.nome + ': resposta fora do formato (' + a.erro + ')' + (t === 1 ? ', tentando de novo' : ''));
  }
  throw new Error(prov.nome + ': formato de auditoria inválido duas vezes');
}

async function produzir(u, ctx, c) {
  let feedback = null; const L = '  [' + u.id + '] ';
  for (let rodada = 1; rodada <= cfg.rodadas_por_unidade; rodada++) {
    const g = await c.g.json({ system: SISTEMA_GERADOR(doutrina), user: pedidoGerador(u, ctx.recentes, feedback), max: 6000, temp: 0.6 });
    const d = montar(g, u);
    const erros = validarLocal(d, { versos: u.verses, existentes: ctx.prontos });
    if (erros.length) { log(L + 'rodada ' + rodada + ': verificação automática recusou: ' + erros.join(', ')); feedback = { anterior: g, falhas: erros.map(x => 'automático: ' + x) }; estado.unidades[u.id].ultimo = 'automático: ' + erros.slice(0, 4).join(', '); continue; }
    const a1 = await auditar(c.a1, u, d);
    if (!a1.aprovado) { log(L + 'rodada ' + rodada + ': ' + c.a1.nome + ' recusou: ' + a1.falhas.join(' | ').slice(0, 400)); feedback = { anterior: g, falhas: a1.falhas.map(x => c.a1.familia + ': ' + x) }; estado.unidades[u.id].ultimo = c.a1.nome + ': ' + a1.falhas.join(' | ').slice(0, 300); continue; }
    const a2 = await auditar(c.a2, u, d);
    if (!a2.aprovado) { log(L + 'rodada ' + rodada + ': ' + c.a2.nome + ' recusou: ' + a2.falhas.join(' | ').slice(0, 400)); feedback = { anterior: g, falhas: a2.falhas.map(x => c.a2.familia + ': ' + x) }; estado.unidades[u.id].ultimo = c.a2.nome + ': ' + a2.falhas.join(' | ').slice(0, 300); continue; }
    d.audit = {
      generator: { model: c.g.modelo, family: c.g.familia },
      auditors: [{ model: c.a1.modelo, family: c.a1.familia }, { model: c.a2.modelo, family: c.a2.familia }],
      gates: Object.fromEntries(GATES.map(x => [x, 'P'])), humanReviewed: false, contractVersion: cfg.contrato, auditedAt: new Date().toISOString(), rounds: rodada

  }
  return null;
}

async function main() {
  const fila = montarFila(R(), cfg);

  let feitos = 0, recusados = 0, falhasSeguidas = 0, pulados = 0, andamento = 0, parar = false, cursor = 0;
  const emUso = new Set();
  log('Fila: ' + fila.length + ' trechos; prontos: ' + idsProntos.size + '; limite desta execução: ' + MAX_ITENS + ' ensinos / ' + MAX_MIN + ' min; ' + CONC + ' em paralelo.');
  function proximo() {
    while (cursor < fila.length) {
      const u = fila[cursor++];
      if (idsProntos.has(u.id) || emUso.has(u.id)) continue;
      const e = estado.unidades[u.id] = estado.unidades[u.id] || { execucoes: 0 };
      if (e.estado === 'travado') { pulados++; continue; }
      const idx = u.cap.findIndex(v => v.n === u.v1), fim = u.cap.findIndex(v => v.n === u.v2);
      u.antes = u.cap.slice(Math.max(0, idx - 3), idx); u.depois = u.cap.slice(fim + 1, fim + 4);

      if (!combo) { log('  Nenhuma combinação válida de gerador e auditores com limite livre; parando. Continua no próximo horário.'); parar = true; return; }

          writeFileSync(join(dir, u.id + '.json'), JSON.stringify(d, null, 1) + '\n');
          prontos.push({ id: d.id, titulo: d.titulo, entender: d.entender, viver: d.viver, ore: d.ore, guarde: d.guarde }); idsProntos.add(d.id);
          e.estado = 'pronto'; e.titulo = d.titulo; feitos++; falhasSeguidas = 0;
          log('  PRONTO [' + u.id + ']: "' + d.titulo + '" (aprovado pelos dois auditores em ' + d.audit.rounds + ' rodada(s))');
        } else {
          e.execucoes++; recusados++; falhasSeguidas = 0;
          if (e.execucoes >= cfg.execucoes_ate_travar) { e.estado = 'travado'; log('  TRAVADO [' + u.id + '] após ' + e.execucoes + ' execuções sem aprovação; precisa de olhar humano: ' + (e.ultimo || '')); }
          else log('  [' + u.id + '] não aprovado nesta execução (' + e.execucoes + '/' + cfg.execucoes_ate_travar + '); tenta de novo depois');

  }

  finalizar(0);
}
